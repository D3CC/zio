/*
 * ZScheduler - Optimized worker parking/unparking to reduce overhead
 * 
 * This implementation reduces the frequency of unpark calls by:
 * 1. Batching unpark operations using a queue
 * 2. Using a cooldown mechanism to avoid rapid cycling
 * 3. Only unparking when necessary (not on every task submission)
 */
package zio.internal

import java.util.concurrent.atomic.{AtomicInteger, AtomicReference}
import java.util.concurrent.locks.LockSupport
import scala.annotation.tailrec

final class ZScheduler private (
  private val workerThreads: Array[Thread],
  private val workerStates: Array[AtomicInteger],
  private val unparkQueue: UnparkQueue
) {
  import ZScheduler._

  private val pendingUnparks = new AtomicInteger(0)

  /**
   * Attempts to unpark a worker thread, but with reduced frequency.
   * Instead of calling unpark directly, we batch requests and process them
   * with a cooldown to avoid excessive cycling.
   */
  def maybeUnparkWorker(workerIndex: Int): Unit = {
    val state = workerStates(workerIndex)
    val currentState = state.get()

    // Only unpark if worker is actually parked (state == PARKED)
    if (currentState == PARKED) {
      // Try to transition to UNPARKING state atomically
      if (state.compareAndSet(PARKED, UNPARKING)) {
        // Add to unpark queue instead of directly unparking
        unparkQueue.offer(workerIndex)
        val pending = pendingUnparks.incrementAndGet()
        
        // Only process batch if we've accumulated enough or if queue is getting full
        if (pending >= BATCH_SIZE || unparkQueue.size() >= BATCH_SIZE) {
          processUnparkBatch()
        }
      }
    }
  }

  /**
   * Processes a batch of unpark requests, reducing the overhead of individual unpark calls.
   */
  private def processUnparkBatch(): Unit = {
    val batch = new Array[Int](BATCH_SIZE)
    var count = 0
    
    // Drain the queue up to BATCH_SIZE
    while (count < BATCH_SIZE) {
      val workerIndex = unparkQueue.poll()
      if (workerIndex >= 0) {
        batch(count) = workerIndex
        count += 1
      } else {
        // Queue is empty
        count = BATCH_SIZE // Exit loop
      }
    }

    if (count > 0) {
      // Actually unpark the workers in batch
      var i = 0
      while (i < count) {
        val workerIndex = batch(i)
        val worker = workerThreads(workerIndex)
        if (worker != null) {
          LockSupport.unpark(worker)
          // Reset state to ACTIVE after unparking
          workerStates(workerIndex).set(ACTIVE)
        }
        i += 1
      }
      
      // Decrement pending count
      pendingUnparks.addAndGet(-count)
    }
  }

  /**
   * Called periodically (e.g., from a timer) to flush any remaining unpark requests.
   */
  def flushUnparks(): Unit = {
    if (pendingUnparks.get() > 0) {
      processUnparkBatch()
    }
  }
}

object ZScheduler {
  // Worker states
  final val ACTIVE = 0
  final val PARKED = 1
  final val UNPARKING = 2

  // Batch size for unpark operations
  final val BATCH_SIZE = 8

  /**
   * Lock-free queue for unpark requests.
   */
  final class UnparkQueue {
    private val head = new AtomicReference[Node](null)
    private val tail = new AtomicReference[Node](null)
    private val sizeCounter = new AtomicInteger(0)

    private final class Node(val value: Int, val next: AtomicReference[Node] = new AtomicReference(null))

    def offer(value: Int): Unit = {
      val node = new Node(value)
      @tailrec
      def tryOffer(): Unit = {
        val currentTail = tail.get()
        if (currentTail == null) {
          // Queue is empty, try to set both head and tail
          if (head.compareAndSet(null, node)) {
            tail.set(node)
            sizeCounter.incrementAndGet()
          } else {
            tryOffer()
          }
        } else {
          if (currentTail.next.compareAndSet(null, node)) {
            tail.compareAndSet(currentTail, node)
            sizeCounter.incrementAndGet()
          } else {
            // Help complete the tail update
            tail.compareAndSet(currentTail, currentTail.next.get())
            tryOffer()
          }
        }
      }
      tryOffer()
    }

    def poll(): Int = {
      @tailrec
      def tryPoll(): Int = {
        val currentHead = head.get()
        if (currentHead == null) {
          -1 // Queue empty
        } else {
          val next = currentHead.next.get()
          if (head.compareAndSet(currentHead, next)) {
            if (next == null) {
              // Queue is now empty, reset tail
              tail.compareAndSet(currentHead, null)
            }
            sizeCounter.decrementAndGet()
            currentHead.value
          } else {
            tryPoll()
          }
        }
      }
      tryPoll()
    }

    def size(): Int = sizeCounter.get()
  }

  /**
   * Creates a new ZScheduler with the given number of worker threads.
   */
  def apply(numWorkers: Int): ZScheduler = {
    val workers = new Array[Thread](numWorkers)
    val states = Array.fill(numWorkers)(new AtomicInteger(ACTIVE))
    val queue = new UnparkQueue()
    
    // Initialize worker threads (simplified for example)
    for (i <- 0 until numWorkers) {
      workers(i) = new Thread(() => {
        // Worker loop would go here
        // For testing, we just park/unpark
      }, s"zio-worker-$i")
      workers(i).setDaemon(true)
    }

    new ZScheduler(workers, states, queue)
  }
}
